cmake_minimum_required(VERSION 3.16)
project(asmodel C)

option(ASMODEL_BUILD_TESTS "Build asmodel tests" ON)
option(ASMODEL_BUILD_FUZZERS "Build Clang libFuzzer parser targets" OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

find_package(CURL QUIET)
add_library(asmodel_obj OBJECT src/asmodel.c src/input.c src/input_json.c src/tools.c src/tool_decode.c src/openai.c src/openai_decode.c src/tokens.c src/pipeline.c
  src/json_support.c src/json_value.c src/json_parse.c src/json_write.c)
target_include_directories(asmodel_obj PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(asmodel_obj PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(asmodel_obj PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(asmodel_obj PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(asmodel_obj PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(asmodel_obj PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(CURL_FOUND)
  target_compile_definitions(asmodel_obj PRIVATE ASMODEL_WITH_CURL)
  target_link_libraries(asmodel_obj PRIVATE CURL::libcurl)
endif()
if(WIN32)
  target_link_libraries(asmodel_obj PRIVATE winhttp)
endif()

add_library(asmodel_static STATIC $<TARGET_OBJECTS:asmodel_obj>)
set_target_properties(asmodel_static PROPERTIES OUTPUT_NAME asmodel)
target_include_directories(asmodel_static PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(asmodel_shared SHARED $<TARGET_OBJECTS:asmodel_obj>)
set_target_properties(asmodel_shared PROPERTIES OUTPUT_NAME asmodel WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(asmodel_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(WIN32)
  set_target_properties(asmodel_shared PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/shared-import)
endif()
if(CURL_FOUND)
  target_link_libraries(asmodel_static PUBLIC CURL::libcurl)
  target_link_libraries(asmodel_shared PUBLIC CURL::libcurl)
endif()
if(WIN32)
  target_link_libraries(asmodel_static PUBLIC winhttp)
  target_link_libraries(asmodel_shared PUBLIC winhttp)
endif()
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(asmodel_static PUBLIC Threads::Threads)
  target_link_libraries(asmodel_shared PUBLIC Threads::Threads)
  target_link_libraries(asmodel_static PUBLIC m)
  target_link_libraries(asmodel_shared PUBLIC m)
endif()

if(ASMODEL_BUILD_FUZZERS)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR WIN32)
    message(FATAL_ERROR "Parser fuzzing requires Clang and a POSIX host")
  endif()
  target_compile_options(asmodel_obj PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  target_link_options(asmodel_static INTERFACE -fsanitize=address,undefined)
  target_link_options(asmodel_shared PRIVATE -fsanitize=address,undefined)
  foreach(name json provider)
    add_executable(asmodel-fuzz-${name} fuzz/${name}.c)
    set_target_properties(asmodel-fuzz-${name} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_include_directories(asmodel-fuzz-${name} PRIVATE src)
    target_link_libraries(asmodel-fuzz-${name} PRIVATE asmodel_static)
    target_compile_options(asmodel-fuzz-${name} PRIVATE -Wall -Wextra -Wpedantic -Werror
      -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(asmodel-fuzz-${name} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()

if(ASMODEL_BUILD_TESTS)
  enable_testing()
  add_executable(asmodel-tools-tests tests/test_tools.c)
  target_include_directories(asmodel-tools-tests PRIVATE src)
  target_link_libraries(asmodel-tools-tests PRIVATE asmodel_static)
  add_test(NAME asmodel-tools COMMAND asmodel-tools-tests)
  add_executable(asmodel-input-tests tests/test_input.c)
  target_include_directories(asmodel-input-tests PRIVATE src)
  target_link_libraries(asmodel-input-tests PRIVATE asmodel_static)
  add_test(NAME asmodel-input COMMAND asmodel-input-tests)
  add_executable(asmodel-embedding-tests tests/test_embedding.c)
  target_link_libraries(asmodel-embedding-tests PRIVATE asmodel_static)
  add_test(NAME asmodel-embedding COMMAND asmodel-embedding-tests)
  add_executable(asmodel-decode-tests tests/test_decode.c)
  target_include_directories(asmodel-decode-tests PRIVATE src)
  target_link_libraries(asmodel-decode-tests PRIVATE asmodel_static)
  add_test(NAME asmodel-decode COMMAND asmodel-decode-tests)
  add_executable(asmodel-tests tests/test_manager.c)
  target_link_libraries(asmodel-tests PRIVATE asmodel_static)
  add_test(NAME asmodel-manager COMMAND asmodel-tests)
  if(NOT WIN32)
    add_executable(asmodel-queue-tests tests/test_queue.c)
    target_link_libraries(asmodel-queue-tests PRIVATE asmodel_static)
    add_test(NAME asmodel-queue COMMAND asmodel-queue-tests)
    set_tests_properties(asmodel-queue PROPERTIES TIMEOUT 5)
  endif()
  if(WIN32 OR CURL_FOUND)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_executable(asmodel-openai-smoke tests/test_openai.c tests/test_messages_wire.c tests/test_tools_wire.c)
      target_link_libraries(asmodel-openai-smoke PRIVATE asmodel_static)
      add_test(NAME asmodel-openai
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_openai.py
                $<TARGET_FILE:asmodel-openai-smoke>)
    endif()
  endif()
endif()
